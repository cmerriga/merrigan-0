   /*
     * namespace StructuredProgramming - describe a program
     *     Function
     *          a Block with parameters
     *          a series
     *          has local named values, including the ones initialized by parameters
     *          has name and pointer to scope where that name means something
     *     Expression
     *          Add
     *          Subtract
     *          Multiply
     *          Equal
     *          Less
     *     Assignment
     *          has variable name
     *          has expression or value
     *     Call
     *          has a series of expressions to evaluate
     *          
     *     Loop
     *     LoopBreak
     *     LoopContinue
     *     FunctionReturn
     *     ParallelFunction
     *     Block
     *          a series
     *          has summary of what it is doing, suitable for logging
     *     Variable
     *          has name
     *          has summary, maybe other annotations like possible or probable values or transitions
     */
Function<int> testFunction = new Function<int>(
    "testFunction",
    "Multiplies a value by 2 until it is no longer less than 123",
    Array.Empty<Variable>(),
    new Statement[] {
        new Assignment<int>("a", 4),
        new Loop(
            new Statement[] {
                new If(
                    new LessThan<int>(new VariableValue<int>("a"), new Literal<int>(123)),
                    new LoopBreak()),
                new Assignment<int>("a", new Multiply<int>(new VariableValue<int>("a"), 2))
            }
        }
        new FunctionReturn<int>(new VariableValue<int>("a"))
    })

    new SetConstantAction<long>("a", 4L),
    new WhileAction(
        new LessThanLongExpression(new GetExpression<long>("a"), 123L),
        new SetExpressionAction<long>(
            "a",
            new MultiplyLongExpression(new GetExpression<long>("a"), 2L))),
    new SetExpressionAction<long>(Execution.ReturnVariableName, new GetExpression<long>("a")));
